# UML-диаграмма прецедентов (Use Case)

> Поведенческая диаграмма: акторы системы и их взаимодействие с функциональностью.

```mermaid
flowchart LR
    subgraph Actors["Акторы"]
        U([👤 Пользователь])
        MOD([🧑‍💼 Модератор])
        SMOD([👨‍💼 Старший модератор])
        DS([👨‍💻 Data Scientist /\nML-инженер])
        SYS([🤖 ML-система\nавтоматический актор])
    end

    subgraph System["Система анализа тональности"]
        subgraph UC_User["Функции для пользователя"]
            UC1["UC-01\nОпубликовать комментарий"]
            UC2["UC-02\nПодать апелляцию\nна блокировку"]
        end

        subgraph UC_Mod["Функции модератора"]
            UC3["UC-03\nПросмотреть очередь\nожидающих проверки"]
            UC4["UC-04\nВынести решение\n(блок / одобрить / эскалировать)"]
            UC5["UC-05\nПросмотреть статистику\nтональности"]
            UC6["UC-06\nРассмотреть апелляцию"]
        end

        subgraph UC_SMod["Функции старшего модератора"]
            UC7["UC-07\nОбработать эскалированный\nкомментарий"]
            UC8["UC-08\nНастроить пороги\nавтоблока"]
            UC9["UC-09\nСформировать отчёт\nдля РКН"]
        end

        subgraph UC_DS["Функции ML-инженера"]
            UC10["UC-10\nЗапустить переобучение\nмодели"]
            UC11["UC-11\nСравнить версии моделей\nв MLflow"]
            UC12["UC-12\nПоставить модель\nна продакшн"]
            UC13["UC-13\nНастроить мониторинг\nдрейфа данных"]
        end

        subgraph UC_Auto["Автоматические функции ML-системы"]
            UC14["UC-14\nАвтоматически классифицировать\nкомментарий"]
            UC15["UC-15\nАвто-заблокировать\nтоксичный комментарий"]
            UC16["UC-16\nПоставить в очередь\nпограничный комментарий"]
            UC17["UC-17\nОтправить алерт\nпри деградации модели"]
        end
    end

    %% Пользователь
    U -->|инициирует| UC1
    U -->|инициирует| UC2

    %% Связи UC-01
    UC1 -.->|"«include»"| UC14
    UC14 -.->|"«extend»"| UC15
    UC14 -.->|"«extend»"| UC16

    %% Модератор
    MOD -->|выполняет| UC3
    MOD -->|выполняет| UC4
    MOD -->|выполняет| UC5
    MOD -->|выполняет| UC6

    UC4 -.->|"«extend»"| UC7

    %% Старший модератор
    SMOD -->|выполняет| UC7
    SMOD -->|выполняет| UC8
    SMOD -->|выполняет| UC9
    SMOD -->|выполняет| UC5

    %% ML-инженер
    DS -->|выполняет| UC10
    DS -->|выполняет| UC11
    DS -->|выполняет| UC12
    DS -->|выполняет| UC13

    %% Автоматические действия
    SYS -->|выполняет| UC14
    SYS -->|выполняет| UC15
    SYS -->|выполняет| UC16
    SYS -->|выполняет| UC17

    style Actors fill:#f8f9fa,stroke:#6c757d
    style UC_User fill:#d4edda,stroke:#28a745
    style UC_Mod fill:#cce5ff,stroke:#004085
    style UC_SMod fill:#fff3cd,stroke:#ffc107
    style UC_DS fill:#f3e5f5,stroke:#7b1fa2
    style UC_Auto fill:#fce4ec,stroke:#c62828
```
